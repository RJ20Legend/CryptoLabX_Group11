Purpose:

def shift_cipher(text, key, mode):
    result = ""

    for char in text:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')

            if mode == 'E':
                shifted = (ord(char) - base + key) % 26
            else:
                shifted = (ord(char) - base - key) % 26

            result += chr(base + shifted)
        else:
            result += char

    return result


print("===================================")
print("       SHIFT CIPHER")
print("===================================")

choice = input("Enter E for Encryption or D for Decryption: ").upper()
text = input("Enter text: ")
key = int(input("Enter key (0-25): "))

if choice == 'E':
    encrypted = shift_cipher(text, key, 'E')
    print("Encrypted Text:", encrypted)

elif choice == 'D':
    decrypted = shift_cipher(text, key, 'D')
    print("Decrypted Text:", decrypted)

else:
    print("Invalid choice!")

import string

def decrypt(text, key):
    result = ""

    for char in text:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')
            result += chr((ord(char) - base - key) % 26 + base)
        else:
            result += char

    return result


def load_dictionary(filename):
    try:
        with open(filename, "r") as file:
            return set(word.strip().upper() for word in file
                       if word.strip())
    except FileNotFoundError:
        print("Dictionary file not found.")
        return set()


def dictionary_score(text, dictionary):
    words = text.upper().split()
    score = 0

    for word in words:
        if word in dictionary:
            score += 1

    return score


print("========================================")
print("      SHIFT CIPHER DICTIONARY ATTACK")
print("========================================")

ciphertext = input("Enter ciphertext: ")

dictionary_path = "attacks/shift_cipher_attack/dictionary/english_words.txt"
dictionary = load_dictionary(dictionary_path)

print("\n========================================")
print("   BRUTE FORCE + DICTIONARY ATTACK")
print("========================================")

print("Ciphertext:", ciphertext)
print("Trying all possible keys...\n")

best_key = 0
best_score = -1
best_plaintext = ""

for key in range(26):
    plaintext = decrypt(ciphertext, key)
    score = dictionary_score(plaintext, dictionary)

    print(f"Key {key:2d} -> Score: {score:2d} -> {plaintext}")

    if score > best_score:
        best_score = score
        best_key = key
        best_plaintext = plaintext

print("\n========================================")
print("              RESULT")
print("========================================")

print("Predicted Key      :", best_key)
print("Dictionary Score   :", best_score)
print("Decrypted Text     :", best_plaintext)

3. chi_square_attack.py
============================================================

Purpose:
Performs Shift Cipher cryptanalysis using English letter-frequency

import string

ENGLISH_FREQ = {
    'A': 8.167, 'B': 1.492, 'C': 2.782, 'D': 4.253,
    'E': 12.702, 'F': 2.228, 'G': 2.015, 'H': 6.094,
    'I': 6.966, 'J': 0.153, 'K': 0.772, 'L': 4.025,
    'M': 2.406, 'N': 6.749, 'O': 7.507, 'P': 1.929,
    'Q': 0.095, 'R': 5.987, 'S': 6.327, 'T': 9.056,
    'U': 2.758, 'V': 0.978, 'W': 2.360, 'X': 0.150,
    'Y': 1.974, 'Z': 0.074
}


def decrypt(text, key):
    result = ""

    for char in text:
        if char.isalpha():
            base = ord('A') if char.isupper() else ord('a')
            result += chr((ord(char) - base - key) % 26 + base)
        else:
            result += char

    return result


def chi_square(text):
    letters = [c.upper() for c in text if c.isalpha()]
    total = len(letters)

    if total == 0:
        return float('inf')

    counts = {letter: 0 for letter in string.ascii_uppercase}

    for letter in letters:
        counts[letter] += 1

    score = 0.0

    for letter in string.ascii_uppercase:
        expected = ENGLISH_FREQ[letter] * total / 100
        observed = counts[letter]

        if expected > 0:
            score += ((observed - expected) ** 2) / expected

    return score


print("========================================")
print("       SHIFT CIPHER CHI-SQUARE ATTACK")
print("========================================")

ciphertext = input("Enter ciphertext: ")

print("\n========================================")
print("       CHI-SQUARE CRYPTANALYSIS")
print("========================================")

print("Ciphertext:", ciphertext)
print("Trying all possible keys...\n")

best_key = 0
best_score = float('inf')
best_plaintext = ""

for key in range(26):
    plaintext = decrypt(ciphertext, key)
    score = chi_square(plaintext)

    print(f"Key {key:2d} -> Chi-Square: {score:9.4f} -> {plaintext}")

    if score < best_score:
        best_score = score
        best_key = key
        best_plaintext = plaintext

print("\n========================================")
print("                RESULT")
print("========================================")

print("Predicted Key     :", best_key)
print("Chi-Square Score  :", f"{best_score:.4f}")
print("Decrypted Text    :", best_plaintext)
